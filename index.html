            <span id="totalHandsDisplay">總手數: 0</span>

        <label for="betAmount" style="color: #aaaaaa; text-align: center;">本局下注金額</label>

        <input type="number" id="betAmount" inputmode="numeric" placeholder="例如: 100">

    </div>


    <div class="btn-group">

        <button class="btn-win" onclick="recordHand('win')">贏</button>

        <button class="btn-push" onclick="recordHand('push')">平手</button>

        <button class="btn-loss" onclick="recordHand('loss')">輸</button>

    </div>


    <button class="btn-reset" onclick="resetData()">清除所有紀錄</button>


    <script>

        // 初始化資料

        let data = JSON.parse(localStorage.getItem('bj_tracker_data')) || {

            netProfit: 0,

            totalHands: 0,

            wins: 0,

            losses: 0,

            pushes: 0

        };


        const betInput = document.getElementById('betAmount');


        // 更新畫面顯示

        function updateUI() {

            const profitEl = document.getElementById('netProfitDisplay');

            profitEl.innerText = (data.netProfit > 0 ? '+' : '') + data.netProfit;

            profitEl.className = 'profit ' + (data.netProfit > 0 ? 'positive' : (data.netProfit < 0 ? 'negative' : ''));

            

            document.getElementById('totalHandsDisplay').innerText = `總手數: ${data.totalHands}`;

            

            let winRate = 0;

            let resolvedHands = data.wins + data.losses; // 計算勝率通常排除平手

            if (resolvedHands > 0) {

                winRate = Math.round((data.wins / resolvedHands) * 100);

            }

            document.getElementById('winRateDisplay').innerText = `勝率: ${winRate}%`;

        }


        // 記錄結果

        function recordHand(outcome) {

            const bet = parseFloat(betInput.value);

            if (isNaN(bet) || bet <= 0) {

                alert("請輸入有效的下注金額！");

                return;

            }


            data.totalHands++;


            if (outcome === 'win') {

                data.netProfit += bet;

                data.wins++;

            } else if (outcome === 'loss') {

                data.netProfit -= bet;

                data.losses++;

            } else if (outcome === 'push') {

                data.pushes++;

            }


            // 存入手機瀏覽器的 LocalStorage

            localStorage.setItem('bj_tracker_data', JSON.stringify(data));

            updateUI();

        }


        // 清除紀錄

        function resetData() {

            if(confirm("確定要清除所有紀錄嗎？")) {

                data = { netProfit: 0, totalHands: 0, wins: 0, losses: 0, pushes: 0 };

                localStorage.removeItem('bj_tracker_data');

                updateUI();

            }

        }


        // 初始載入時更新一次畫面

        updateUI();

    </script>
